cmake_minimum_required(VERSION 3.20)
project(genehub_capture_engine LANGUAGES C CXX)
if(APPLE)
  # Enable in this directory before the native subproject creates ObjC++
  # sources; enabling it only in a child leaves parent rule variables unset.
  enable_language(OBJCXX)
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(MW_VERSION "genehub-capture-0.1")
set(MW_NATIVE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/.cache/mw-native-host" CACHE PATH "Pinned standalone native module")
if(NOT EXISTS "${MW_NATIVE_SOURCE}/CMakeLists.txt")
  message(FATAL_ERROR "Run python3 prepare.py first; see apps/capture/README.md")
endif()
add_subdirectory("${MW_NATIVE_SOURCE}" mw-native-host)
if(APPLE)
  target_sources(mw-native-host PRIVATE "${MW_NATIVE_SOURCE}/src/platform/macos/GeneHubInput.mm")
  target_link_libraries(mw-native-host PRIVATE "-framework ColorSync")
  set_source_files_properties("${MW_NATIVE_SOURCE}/src/platform/macos/GeneHubInput.mm" PROPERTIES COMPILE_FLAGS "-fobjc-arc")
endif()
add_library(genehub-capture-engine SHARED capture.cpp)
if(APPLE)
  target_sources(genehub-capture-engine PRIVATE manager_mac.mm)
  set_source_files_properties(manager_mac.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
  target_link_libraries(genehub-capture-engine PRIVATE "-framework Cocoa" "-framework ApplicationServices")
elseif(WIN32)
  target_sources(genehub-capture-engine PRIVATE manager_win.cpp)
  target_link_libraries(genehub-capture-engine PRIVATE user32 gdi32 wtsapi32)
else()
  target_sources(genehub-capture-engine PRIVATE manager_none.cpp)
endif()
if(MSVC)
  target_compile_options(mw-native-host PRIVATE /utf-8)
  target_compile_options(genehub-capture-engine PRIVATE /utf-8)
endif()
if(WIN32)
  target_compile_definitions(genehub-capture-engine PRIVATE UNICODE _UNICODE)
  target_include_directories(genehub-capture-engine PRIVATE "${MW_NATIVE_SOURCE}/src")
  # The pinned module also compiles its optional D3D12 paths. Use a pinned
  # header set for those declarations instead of requiring a system SDK
  # upgrade. Capture still explicitly selects the established D3D11 path.
  target_include_directories(mw-native-host BEFORE PRIVATE "${MW_NATIVE_SOURCE}/third_party/directx-headers/include/directx")
  target_include_directories(genehub-capture-engine BEFORE PRIVATE "${MW_NATIVE_SOURCE}/third_party/directx-headers/include/directx")
endif()
target_compile_features(genehub-capture-engine PRIVATE cxx_std_17)
target_link_libraries(genehub-capture-engine PRIVATE mw-native-host)
set_target_properties(genehub-capture-engine PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
install(TARGETS genehub-capture-engine RUNTIME DESTINATION . LIBRARY DESTINATION .)
